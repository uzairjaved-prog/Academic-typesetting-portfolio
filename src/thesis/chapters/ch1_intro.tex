\chapter{Introduction}\label{ch:intro}
\section{Motivation}
Energy monitoring\index{energy monitoring} systems produce long hourly series whose behaviour is dominated by weather. Understanding the temperature--demand\index{demand model} link supports forecasting and control~\cite{ogata2010modern}.

\section{Research Questions}
\begin{enumerate}[label=\textbf{RQ\arabic*.}]
  \item How strongly is power demand coupled to temperature?
  \item Do monitoring sites differ systematically?
  \item Can a feedback controller regulate the resulting demand?
\end{enumerate}
