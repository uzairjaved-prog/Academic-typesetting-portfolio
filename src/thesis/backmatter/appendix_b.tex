\chapter{Derivation of the OLS Covariance}\label{app:deriv}

Substituting $\bm y=\bm X\bm\beta+\bm\varepsilon$ into \cref{eq:thesis-ols} gives
\[
  \hat{\bm\beta}
  =\bm\beta+(\bm X^{\top}\bm X)^{-1}\bm X^{\top}\bm\varepsilon.
\]
If $\E[\bm\varepsilon]=\bm 0$ and $\Cov(\bm\varepsilon)=\sigma^2\bm I$, then
\begin{align}
  \E[\hat{\bm\beta}] &= \bm\beta,\\
  \Cov(\hat{\bm\beta})
    &= (\bm X^{\top}\bm X)^{-1}\bm X^{\top}
       (\sigma^2\bm I)\bm X(\bm X^{\top}\bm X)^{-1}\\
    &= \sigma^2(\bm X^{\top}\bm X)^{-1}.
\end{align}
